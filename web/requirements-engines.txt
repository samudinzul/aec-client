# Extra-engine set — optional backends (WebRTC AEC3, NKF-AEC).
# Neither ships in a base install; the chain fails closed with a
# clear error if a profile is picked without its package, and the
# UI disables options whose backend is missing. (WebRTC AEC3 is a
# regular listed choice now, not experimental-gated — only NKF-AEC
# hides behind the flag. The package itself stays optional.)
# Install:  pip install -r web\requirements-engines.txt

                      # prebuilt wheels incl. Windows x86_64, Python 3.10-3.14
onnxruntime>=1.20     # NKF-AEC engine: runs models/nkf.onnx on CPU
                      # (Windows wheels; 45 KB model ships in this repo).
                      # Also runs the Hush NS backend (web/hush_onnx.py:
                      # enc/erb_dec/df_dec from the shipped bundle, no
                      # DLL needed) — same fail-closed gating as NKF.
onnx>=1.17            # Hush GRU-carry discovery only: inspects the
                      # bundle's graph once at load (recurrent node
                      # state names) — lazy import, never in the base
                      # set. Mainstream ONNX-project wheel.
